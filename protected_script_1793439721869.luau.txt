-- Tạo giao diện chính
local ScreenGui = Instance.new("ScreenGui")
ScreenGui.Name = "ChilliVS"
ScreenGui.ResetOnSpawn = false
ScreenGui.Parent = game:GetService("CoreGui") -- Nếu lỗi hãy đổi thành game:GetService("Players").LocalPlayer:WaitForChild("PlayerGui")

-- Hàm tạo Khung (Panel) cho từng bên
local function CreatePanel(name, posX, titleText, themeColor)
    -- Sử dụng TextButton để toàn bộ khung có thể click được
    local PanelBtn = Instance.new("TextButton")
    PanelBtn.Name = name
    PanelBtn.Size = UDim2.new(0.4, 0, 0.6, 0) -- Chiếm 40% màn hình
    PanelBtn.Position = UDim2.new(posX, 0, 0.2, 0)
    
    -- Làm trong suốt khung
    PanelBtn.BackgroundColor3 = Color3.fromRGB(15, 15, 15)
    PanelBtn.BackgroundTransparency = 0.5 
    
    PanelBtn.BorderSizePixel = 2
    PanelBtn.BorderColor3 = themeColor
    PanelBtn.Text = "" 
    PanelBtn.AutoButtonColor = false 
    PanelBtn.Parent = ScreenGui 

    -- Bo góc
    local UICorner = Instance.new("UICorner")
    UICorner.CornerRadius = UDim.new(0, 10)
    UICorner.Parent = PanelBtn

    -- Hiệu ứng viền phát sáng
    local UIStroke = Instance.new("UIStroke")
    UIStroke.Color = themeColor
    UIStroke.Thickness = 2
    UIStroke.Transparency = 0.2
    UIStroke.Parent = PanelBtn

    -- Tiêu đề lớn (Chilli v8 / Chilli v8 Pro) - Căn chính giữa khung
    local Title = Instance.new("TextLabel")
    Title.Size = UDim2.new(1, 0, 0.3, 0) 
    Title.Position = UDim2.new(0, 0, 0.5, 0) -- Đưa ra giữa
    Title.AnchorPoint = Vector2.new(0, 0.5) -- Căn tâm chuẩn xác
    Title.BackgroundTransparency = 1
    Title.Text = titleText
    Title.TextColor3 = themeColor
    Title.Font = Enum.Font.GothamBold
    Title.TextScaled = true
    Title.Parent = PanelBtn

    -- Hiệu ứng Hover cho toàn bộ khung
    PanelBtn.MouseEnter:Connect(function()
        UIStroke.Transparency = 0
        UIStroke.Thickness = 4
        PanelBtn.BackgroundTransparency = 0.2 
    end)
    PanelBtn.MouseLeave:Connect(function()
        UIStroke.Transparency = 0.2
        UIStroke.Thickness = 2
        PanelBtn.BackgroundTransparency = 0.5 
    end)

    return PanelBtn
end

-- Chữ V (Đỏ) và S (Xanh) ở giữa sử dụng RichText
local VS = Instance.new("TextLabel")
VS.Size = UDim2.new(0.1, 0, 0.2, 0)
VS.Position = UDim2.new(0.45, 0, 0.4, 0)
VS.BackgroundTransparency = 1
VS.RichText = true -- Bật tính năng RichText để đổi màu từng chữ
-- Chữ V màu Đỏ, chữ S màu Xanh lam
VS.Text = '<font color="rgb(255, 50, 50)">V</font><font color="rgb(0, 150, 255)">S</font>' 
VS.Font = Enum.Font.GothamBlack
VS.TextScaled = true
VS.Parent = ScreenGui

-- Code của 2 bên (Chạy ngầm)
local codeLeft = 'loadstring(game:HttpGet("https://raw.githubusercontent.com/napun87/stealanegg/refs/heads/main/chilli.otcaydodaiNoKey3.7.lua"))()'
local codeRight = 'loadstring(game:HttpGet("https://raw.githubusercontent.com/napun67/stealanegg/refs/heads/main/chillihubv8.lua"))()'

-- Tạo 2 Panel (Trái: Xanh lam / Phải: Đỏ)
local BtnLeft = CreatePanel("LeftPanel", 0.05, "CHILLI V8", Color3.fromRGB(0, 150, 255))
local BtnRight = CreatePanel("RightPanel", 0.55, "CHILLI V8 PRO", Color3.fromRGB(255, 50, 50))

-- HÀM XỬ LÝ KHI BẤM CHỌN
local function OnSelect(scriptCode)
    -- 1. Đóng giao diện ngay lập tức
    ScreenGui:Destroy()
    
    -- 2. Chạy đoạn code
    local success, errorMessage = pcall(function()
        loadstring(scriptCode)()
    end)
    
    if not success then
        warn("Lỗi khi chạy script: " .. tostring(errorMessage))
    end
end

-- Gắn sự kiện Click
BtnLeft.MouseButton1Click:Connect(function()
    OnSelect(codeLeft)
end)

BtnRight.MouseButton1Click:Connect(function()
    OnSelect(codeRight)
end)
